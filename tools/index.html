#!/usr/bin/env bash # xen-pages-put — publish ONE file to a Pages repo through the GitHub Contents API. # # WHY THIS EXISTS (measured, 2026-08-20): the receipts publisher pushes hitthe.link # roughly every 90s while a Pages build takes ~150s. Those numbers cannot converge — # each push cancels the in-flight build, which is the `dur=0ms` "errored" pattern in # logs/pages-watch.log (9 errored / 105 built over 7 days, ZERO real faults). The # Contents API takes a PER-FILE blob sha, so it never fights the commit graph. # # TRAP THIS FIXES: `gh api --jq .sha` on a MISSING file does not return empty # — it can emit the error body on stdout, so a naive capture sets sha='{"message":...}' # and the next update carries a bogus sha. Probe the STATUS, never the piped body. # # usage: xen-pages-put [commit message] set -euo pipefail REPO=${1:?repo, e.g. xlrdtech/hitthe.link} RPATH=${2:?path in repo, e.g. tools/index.html} LOCAL=${3:?local file} MSG=${4:-"publish: $RPATH"} [ -f "$LOCAL" ] || { echo "no such file: $LOCAL" >&2; exit 2; } # STATUS first — the only reliable create-vs-update signal. code=$(gh api "repos/$REPO/contents/$RPATH" --include --silent 2>/dev/null \ | head -1 | awk '{print $2}') || code="" sha="" if [ "$code" = "200" ]; then sha=$(gh api "repos/$REPO/contents/$RPATH" --jq '.sha' 2>/dev/null || true) case "$sha" in *'{'*|*' '*|'') sha="" ;; esac # reject anything not a bare hex sha fi [ -n "$sha" ] && echo "update (blob ${sha:0:12})" || echo "create (no existing blob, status=${code:-none})" args=(-X PUT "repos/$REPO/contents/$RPATH" -f "message=$MSG" -f "content=$(base64 -i "$LOCAL" | tr -d '\n')" -f branch=main) [ -n "$sha" ] && args+=(-f "sha=$sha") gh api "${args[@]}" --jq '"commit \(.commit.sha[0:12]) \(.content.path) \(.content.size)B"'
